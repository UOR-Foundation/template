  proof: induction{
    branches: [
      {proof: simplify{definitions: [ayes, cast, nays]}, binders: [], constructor: nil},

        proof: cases{
          branches: [
            {
              proof: linear_arithmetic{definitions: [ayes, cast, nays]},
              binders: [],
              constructor: aye

            {
              proof: linear_arithmetic{definitions: [ayes, cast, nays]},
              binders: [],
              constructor: nay
